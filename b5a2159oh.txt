    49	export function parseCellAddress(addr: string): [number, number] {
    50	  const m = String(addr).trim().toUpperCase().match(/^([A-Z]+)([1-9]\d*)$/);
    51	  if (!m) throw new Error(`Invalid cell address: ${addr}`);
    52	  let col = 0;
    53	  for (let i = 0; i < m[1].length; i++) {
    54	    col = col * 26 + (m[1].charCodeAt(i) - 64);
    55	  }
    56	  return [parseInt(m[2], 10) - 1, col - 1];
    57	}

   102	              textCount: ['text', 'count'],

   124	              theme: ['theme'],
   125	              slideOrder: ['titles'],
   126	              action: ['action'],
   127	            };

   129	      const goalType = String(g['type']);

   133	          throw new Error(`${goalType}.${key} is required.`);
   134	        }
   135	      }
   136	
   137	      for (const key of [
   138	        'text',
   139	        'action',

   142	        'layout',
   143	        'theme',
   144	        'cell',
   145	        'range',

   147	        if (g[key] !== undefined && typeof g[key] !== 'string') {
   148	          throw new Error(`${goalType}.${key} must be text.`);
   149	        }
   150	      }
   151	
   152	      for (const key of [
   153	        'items',
   154	        'sequence',
   155	        'titles',
   156	        'notContains',
   157	        'allowed',
   158	      ]) {
   159	        if (
   160	          g[key] !== undefined &&
   161	          (!Array.isArray(g[key]) || g[key].some((v: unknown) => typeof v !== 'string'))
   162	        ) {
   163	          throw new Error(`${goalType}.${key} must be a list of strings.`);
   164	        }
   165	      }
   166	
   167	      for (const key of [
   168	        'count',
   169	        'min',
   170	        'minItems',
   171	        'minLines',
   172	        'rows',
   173	        'columns',
   174	        'slide',
   175	      ]) {
   176	        if (g[key] !== undefined && (!Number.isInteger(g[key]) || g[key] < 0)) {
   177	          throw new Error(`${goalType}.${key} must be a non-negative integer.`);
   178	        }
   179	      }
   180	
   181	      if (g['slide'] !== undefined && g['slide'] < 1) {
   182	        throw new Error('Slide numbers start at 1.');
   183	      }
   184	
   185	      if (kind === 'excel') {
   186	        for (const key of ['cell', 'range']) {
   187	          if (g[key]) validateCellRange(g[key]);
   188	        }
   189	        if (goalType === 'cellValue') {
   190	          const cells = g['cells'];
   191	          if (typeof cells !== 'object' || Array.isArray(cells) || !cells) {
   192	            throw new Error('cellValue.cells must be an object.');

   195	        }
   196	      }
   197	    }
   198	  };
   199	
   200	  if (kind === 'word' && !act['tasks']) {
   201	    if (
   202	      typeof act['instruction'] !== 'string' ||
   203	      typeof act['startingDocument'] !== 'string'
   204	    ) {
   205	      throw new Error('instruction and startingDocument are required.');
   206	    }
   207	    validateGoals(act['goals'] || []);
   208	  } else {
   209	    if (
   210	      !['lesson', 'assessment'].includes(act['mode']) ||
   211	      typeof act['title'] !== 'string' ||
   212	      !Array.isArray(act['tasks']) ||
   213	      !act['tasks'].length
   214	    ) {
   215	      throw new Error('mode, title and tasks are required.');
   216	    }
   217	
   218	    if (
   219	      (act['mode'] === 'lesson' && act['tasks'].length !== 1) ||
   220	      (act['mode'] === 'assessment' && act['tasks'].length < 2)
   221	    ) {
   222	      throw new Error(
   223	        'A lesson needs one task; an assessment needs at least two.'
   224	      );
   225	    }
   226

   229	        typeof t['title'] !== 'string' ||
   230	        typeof t['explanation'] !== 'string' ||
   231	        !Array.isArray(t['instructions']) ||
   232	        t['instructions'].length < 1 ||
   233	        t['instructions'].length > 3 ||
   234	        t['instructions'].some((x: unknown) => typeof x !== 'string')
   235	      ) {
   236	        throw new Error(
   237	          'Each task needs a title, explanation and 1–3 instructions.'
   238	        );
   239	      }
   240	
   241	      if (t['showMeHow'] !== undefined && typeof t['showMeHow'] !== 'boolean') {
   242	        throw new Error('showMeHow must be a boolean.');
   243	      }
   244	      validateGoals(t['goals'] || []);
   245	
   246	      if (kind === 'word') {
   247	        const s = t['startingDocument'];
   248	        if (s === null && i > 0) return;
   249	        if (typeof s !== 'string') {
   250	          throw new Error('startingDocument is required for the first task.');
   251	        }
   252	      } else {
   253	        const key = kind === 'excel' ? 'startingSheet' : 'startingDeck';
   254	        const s = t[key];
   255	        if (s === null && i > 0) return;
   256	        if (!s || typeof s !== 'object') {
   257	          throw new Error(`${key} is required for the first task.`);
   258	        }
   259	
   260	        if (
   261	          kind === 'excel' &&
   262	          (!s.cells || typeof s.cells !== 'object' || Array.isArray(s.cells))
   263	        ) {
   264	          throw new Error('startingSheet.cells must be an object.');
   265	        }
   266	
   267	        if (
   268	          kind === 'powerpoint' &&
   269	          (!Array.isArray(s.slides) || !s.slides.length)
   270	        ) {
   271	          throw new Error('startingDeck needs at least one slide.');
   272	        }
   273	      }
   274	    });
   275	  }
   276	
   277	  return act as ActivityPayload;

   114	    if (!this.editorElement) return false;
   115	    const root = this.editorElement;
   116	    const p2 = this.page2Element;
   117	    const text = (root.textContent || '') + (p2 ? ' ' + (p2.textContent || '') : '');

   141	      case 'paragraph': {

   312	                rep.classList.add('doc-title');
   313	              } else if (g['style'] === 'subtitle') {
   314	                rep.classList.add('doc-subtitle');
   315	              }
   316	              b.replaceWith(rep);
   317	            }
   318	          }
   319	        }
   320	        break;
   321	      }
   322	
   323	      case 'list': {
   324	        const ul = document.createElement(g['listType'] === 'bullet' ? 'ul' : 'ol');
   325	        (g['items'] || []).forEach((it: string) => {
   326	          const li = document.createElement('li');
   327	          li.textContent = it;
   328	          ul.append(li);
   329	        });
   330	        root.append(ul);
   331	        break;
   332	      }
   333	
   334	      case 'table': {
   335	        const rows = g['rows'] || 2;
   336	        const cols = g['columns'] || 2;
   337	        const tbl = document.createElement('table');
   338	        tbl.className = 'doc-table';
   339	        for (let i = 0; i < rows; i++) {
   340	          const tr = tbl.insertRow();
   341	          for (let j = 0; j < cols; j++) {
   342	            const td = tr.insertCell();
   343	            td.textContent = g['cells']?.[i]?.[j] || 'Cell';
   344	          }
   345	        }
   346	        root.append(tbl);
   347	        break;
   348	      }
   349	
   350	      case 'image': {
   351	        const img = document.createElement('img');
   352	        img.src = safeAsset(activity?.assetBaseUrl || '', g['file'] || 'photo.png');
   353	        img.dataset['file'] = g['file'];
   354	        if (g['align']) img.dataset['align'] = g['align'];
   355	        if (g['wrap']) img.dataset['wrap'] = g['wrap'];
   356	        if (g['minWidth']) img.width = g['minWidth'];
   357	        root.append(img);
   358	        break;
   359	      }
   360	
   361	      case 'pageSetup': {
   362	        if (g['orientation']) this.setup.orientation = g['orientation'];
   363	        if (g['paperSize']) this.setup.paperSize = g['paperSize'];
   364	        if (g['margins']) Object.assign(this.setup.margins, g['margins']);
   365	        break;
   366	      }
   367	
   368	      case 'element': {
   369	        if (g['element'] === 'header') this.header = g['text'] || 'Header';
   370	        else if (g['element'] === 'footer') this.footer = g['text'] || 'Footer';
   371	        else if (g['element'] === 'pageNumbers') this.numbering = true;
   372	        else if (g['element'] === 'tableOfContents') {
   373	          const toc = document.createElement('div');
   374	          toc.className = 'table-of-contents';
   375	          toc.dataset['element'] = 'tableOfContents';
   376	          toc.innerHTML = '<h3 class="toc-title">Table of Contents</h3><ul class="toc-list"><li>Heading 1</li></ul>';
   377	          root.prepend(toc);
   378	        } else if (g['element'] === 'footnote') {
   379	          const fn = document.createElement('div');
   380	          fn.className = 'footnote';
   381	          fn.dataset['element'] = 'footnote';
   382	          fn.innerHTML = '<hr><p><sup>1</sup> Footnote text</p>';
   383	          root.append(fn);
   384	        } else {
   385	          const el = document.createElement('div');
   386	          el.dataset['element'] = g['element'];
   387	          el.textContent = g['text'] || g['element'];
   388	          root.append(el);
   389	        }
   390	        break;
   391	      }
   392	
   393	      case 'textContains': {
   394	        let currentHtml = root.innerHTML;
   395	        const notList = (g['notContains'] || []) as string[];
   396	        const conList = (g['contains'] || []) as string[];
   397	        notList.forEach((wrong, idx) => {
   398	          const replacement = conList[idx] || conList[0] || '';
   399	          if (wrong && replacement) {
   400	            currentHtml = currentHtml.split(wrong).join(replacement);
   401	          }
   402	        });
   403	        root.innerHTML = currentHtml;
   404	        const text = root.textContent || '';
   405	        conList.forEach((req) => {
   406	          if (!text.includes(req)) {
   407	            const p = document.createElement('p');
   408	            p.textContent = req;
   409	            root.append(p);
   410	          }
   411	        });
   412	        break;
   413	      }
   414	
   415	      case 'textCount': {
   416	        const target = g['text'];
   417	        const needed = g['count'];
   418	        if (target && typeof needed === 'number') {
   419	          let text = root.textContent || '';
   420	          let currentCount = text.split(target).length - 1;
   421	          while (currentCount < needed) {
   422	            const p = document.createElement('p');
   423	            p.textContent = target;
   424	            root.append(p);
   425	            currentCount++;
   426	          }
   427	          if (currentCount > needed) {
   428	            let html = root.innerHTML;
   429	            while (currentCount > needed && html.includes(target)) {
   430	              html = html.replace(target, '');
   431	              currentCount--;
   432	            }
   433	            root.innerHTML = html;
   434	          }
   435	        }
   436	        break;
   437	      }
   438	
   439	      case 'textOrder': {
   440	        const seq = (g['sequence'] || []) as string[];
   441	        if (seq.length > 0) {
   442	          const p = document.createElement('p');
   443	          p.textContent = seq.join(' ');
   444	          root.append(p);
   445	        }
   446	        break;
   447	      }
   448	
   449	      case 'action':
   450	        this.logAction(g['action'], g['text']);
   451	        break;
   452	    }
   453	  }
   454	
   455	  snapshot(): Record<string, any> {
   456	    return {
   457	      html: this.editorElement ? this.editorElement.innerHTML : this.html,
   458	      setup: clone(this.setup),
   459	      header: this.header,
   460	      footer: this.footer,
   461	      numbering: this.numbering,
   462	      hasPageBreak: this.hasPageBreak,
   463	      page2Html: this.page2Element ? this.page2Element.innerHTML : '',
   464	      comments: clone(this.comments),
   465	      styles: clone(this.styles),
   466	      actions: clone(this.actions),
   467	    };
   468	  }
   469	
   470	  restore(s: Record<string, any>): void {
   471	    this.html = s['html'] || '';
   472	    if (this.editorElement) {
   473	      this.editorElement.innerHTML = this.html;
   474	    }
   475	    this.hasPageBreak = !!s['hasPageBreak'];
   476	    if (this.page2Element) {
   477	      this.page2Element.innerHTML = s['page2Html'] || '';
   478	    }
   479	    this.setup = clone(s['setup']);
   480	    this.header = s['header'] || '';
   481	    this.footer = s['footer'] || '';
   482	    this.numbering = !!s['numbering'];
   483	    this.comments = clone(s['comments'] || []);
   484	    this.styles = clone(s['styles'] || {});
   485	    this.actions = clone(s['actions'] || []);
   486	  }
